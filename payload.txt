FIREBASE_URL = "https://ff-bind-system-default-rtdb.firebaseio.com"
CURRENT_VERSION = "2"  # Admin panel se match hona chahiye update ke liye

class Security:
    def __init__(self):
        self.base_url = FIREBASE_URL
        self.key_file = "license.key"

    def check_server(self):
        """Checks Status and Updates"""
        try:
            # Config Check
            r = requests.get(f"{self.base_url}/config.json")
            data = r.json()

            if not data: return

            # 1. Maintenance Check
            if data.get('status') != 'active':
                print(f"\n{Fore.RED}╔{'═'*60}╗")
                print(f"{Fore.RED}║     TOOL IS CURRENTLY OFFLINE / UNDER MAINTENANCE      ║")
                print(f"{Fore.RED}╚{'═'*60}╝")
                print(f"\n{Fore.YELLOW}Message from Admin: {Fore.WHITE}{data.get('message', 'Wait for update.')}")
                input(f"\n{Fore.CYAN}Press Enter to Exit...")
                sys.exit()

            # 2. Auto Update Check
            server_ver = str(data.get('version', '1.0'))
            if server_ver != CURRENT_VERSION:
                print(f"\n{Fore.GREEN}[UPDATE] New version found: v{server_ver}")
                print(f"{Fore.YELLOW}Downloading update automatically...")
                
                update_url = data.get('url')
                if update_url:
                    try:
                        new_code = requests.get(update_url).text
                        if "import sys" in new_code: # Simple integrity check
                            with open(sys.argv[0], 'w', encoding='utf-8') as f:
                                f.write(new_code)
                            print(f"{Fore.GREEN}Update Installed Successfully! Please restart the tool.")
                            sys.exit()
                        else:
                            print(f"{Fore.RED}[ERROR] Downloaded file seems corrupted.")
                    except Exception as e:
                        print(f"{Fore.RED}[ERROR] Update failed: {e}")
                
        except Exception as e:
            # Agar internet nahi chala to skip karke login par jayega
            # Lekin agar strict security chahiye to yaha exit kar sakte ho
            pass

    def validate_key(self, key):
        try:
            r = requests.get(f"{self.base_url}/users/{key}.json")
            data = r.json()
            
            if not data: return False, "Invalid Key"
            
            # Check Expiry
            expiry_str = data.get('expiry')
            if not expiry_str: return False, "No Expiry Date Found"
            
            expiry_date = datetime.datetime.strptime(expiry_str, "%Y-%m-%d")
            if datetime.datetime.now() > expiry_date:
                return False, f"Key Expired on {expiry_str}"
            
            # Check Status (Block/Unblock)
            if data.get('status') != 'active':
                return False, "Key has been Blocked by Admin"

            return True, f"Expires: {expiry_str}"
        except:
            return False, "Connection Error"

    def login(self):
        self.check_server()
        
        # Check Saved Key
        if os.path.exists(self.key_file):
            with open(self.key_file, 'r') as f:
                saved_key = f.read().strip()
            
            print(f"{Fore.CYAN}[SYSTEM] Checking saved license...")
            is_valid, msg = self.validate_key(saved_key)
            if is_valid:
                print(f"{Fore.GREEN}[SUCCESS] Login Successful! ({msg})")
                time.sleep(1)
                return
            else:
                print(f"{Fore.RED}[ERROR] {msg}")
                os.remove(self.key_file) # Delete invalid key

        # Ask for New Key
        while True:
            print(f"\n{Fore.YELLOW}════════════ LICENSE VERIFICATION ════════════")
            user_key = input(f"{Fore.CYAN}Enter License Key: {Fore.WHITE}").strip()
            
            if not user_key: continue

            print(f"{Fore.YELLOW}Verifying...")
            is_valid, msg = self.validate_key(user_key)
            
            if is_valid:
                with open(self.key_file, 'w') as f:
                    f.write(user_key)
                print(f"{Fore.GREEN}[SUCCESS] Access Granted! ({msg})")
                time.sleep(1.5)
                break
            else:
                print(f"{Fore.RED}[FAILED] {msg}")

# --- LOADER ---
class Loader:
    def __init__(self, desc="Loading...", end="Done!", timeout=0.1):
        self.desc = desc
        self.end = end
        self.timeout = timeout
        self._thread = Thread(target=self._animate, daemon=True)
        self.steps = ["|", "/", "-", "\\"]
        self.done = False

    def start(self):
        self._thread.start()
        return self

    def _animate(self):
        from itertools import cycle
        for c in cycle(self.steps):
            if self.done:
                break
            print(f"\r{Colors.YELLOW}{self.desc} {Colors.WHITE}{c}", end="\r")
            time.sleep(self.timeout)

    def stop(self):
        self.done = True
        print(f"\r{Colors.GREEN}✓ {self.desc} ... {self.end}" + " " * 10)

        # agar line zyada lambi ho terminal se, truncate kar do
        if len(line) > inner:
            line = line[:inner]
        print(f"{color}║ {line.center(inner)} ║")
    print(f"{color}╚{'═' * (width - 2)}╝")

class GarenaManager:

    def set_token(self, token):

    def _parse_error(self, result):
        error_code = str(result.get("error", ""))
        res_code = result.get("result")
        if "error_token" in error_code or res_code == 1003:
            return "The Access Token is Invalid or Expired."
        elif "error_password" in error_code:
            return "The Secondary Password is Wrong."
        elif "error_otp" in error_code:
            return "The OTP Code is Wrong or Expired."
        elif "error_email_format" in error_code:
            return "Invalid Email Format."
        elif res_code == 1005:
            return "No Pending Bind Request Found to Cancel."
        elif res_code == 1006:
            return "The Bind Request has Expired."
        elif res_code == 6:
            return "Too Many Requests (Rate Limit). Try again later."
        else:
            return f"Server Error: {error_code} (Code: {res_code})"

    def _convert_seconds(self, s):
        d, h = divmod(s, 86400)
        h, m = divmod(h, 3600)
        m, s = divmod(m, 60)
        return f"{d} Day {h} Hour {m} Min {s} Sec"

    # --- SEND OTP ---
    def send_otp(self, email):
        url = f"{self.base_url}/game/account_security/bind:send_otp"
        data = {"app_id": self.app_id, "access_token": self.access_token,
                "email": email, "locale": "en_US"}
        try:
            load = Loader("Sending OTP to email").start()
            response = self.session.post(url, data=data)
            load.stop()
            result = response.json()
            if response.status_code == 200 and result.get("result") == 0:
                return True, "OTP sent successfully!"
            return False, self._parse_error(result)
        except Exception as e:
            if "Max retries" in str(e) or "NameResolutionError" in str(e) or "Failed to resolve" in str(e):
                return False, "Connection Failed: Server Unreachable.\n     Please check your Internet Connection."
            return False, str(e)

    # --- CONFIRM BIND ---
    def confirm_bind(self, email, otp, password):
        try:
            v_res = self.session.post(
                f"{self.base_url}/game/account_security/bind:verify_otp",
                data={"app_id": self.app_id, "access_token": self.access_token,
                      "otp": otp, "email": email})
            v_result = v_res.json()
            if v_result.get("result") != 0:
                return False, self._parse_error(v_result)
            verifier_token = v_result.get("verifier_token")
            b_res = self.session.post(
                f"{self.base_url}/game/account_security/bind:create_bind_request",
                data={"app_id": self.app_id, "access_token": self.access_token,
                      "verifier_token": verifier_token,
                      "secondary_password": password, "email": email})
            b_result = b_res.json()
            if b_result.get("result") == 0:
                return True, "Account successfully bound!"
            return False, self._parse_error(b_result)
        except Exception as e:
            if "Max retries" in str(e) or "NameResolutionError" in str(e) or "Failed to resolve" in str(e):
                return False, "Connection Failed: Server Unreachable.\n     Please check your Internet Connection."
            return False, str(e)

    # --- CANCEL BIND ---
    def cancel_bind(self):
        try:
            response = self.session.post(
                f"{self.base_url}/game/account_security/bind:cancel_request",
                data={"app_id": self.app_id, "access_token": self.access_token})
            result = response.json()
            if result.get("result") == 0:
                return True, "Bind request cancelled successfully."
            return False, self._parse_error(result)
        except Exception as e:
            if "Max retries" in str(e) or "NameResolutionError" in str(e) or "Failed to resolve" in str(e):
                return False, "Connection Failed: Server Unreachable.\n     Please check your Internet Connection."
            return False, str(e)

    # --- UNBIND ACCOUNT ---
    def unbind_account(self, password):
        try:
            v_res = self.session.post(
                f"{self.base_url}/game/account_security/bind:verify_identity",
                data={"app_id": self.app_id, "access_token": self.access_token,
                      "secondary_password": password})
            v_result = v_res.json()
            if v_result.get("result") != 0:
                return False, self._parse_error(v_result)
            identity_token = v_result.get("identity_token")
            u_res = self.session.post(
                f"{self.base_url}/game/account_security/bind:create_unbind_request",
                data={"app_id": self.app_id, "access_token": self.access_token,
                      "identity_token": identity_token})
            u_result = u_res.json()
            if u_result.get("result") == 0:
                return True, "Unbind process started (15 Days waiting period)."
            return False, self._parse_error(u_result)
        except Exception as e:
            if "Max retries" in str(e) or "NameResolutionError" in str(e) or "Failed to resolve" in str(e):
                return False, "Connection Failed: Server Unreachable.\n     Please check your Internet Connection."
            return False, str(e)
